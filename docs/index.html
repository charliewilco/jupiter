Metis

A source-driven theme family for the editors, terminals, and renderers this repo ships.

Preview

A compact palette with enough separation for code, terminals, and renderers.

Metis keeps the palette small and maps each accent through shared role definitions. The same source file produces light and dark presets, keeping the rendered code samples, editor themes, and terminal colors in lockstep.

type Output = "ghostty" | "xcode" | "prism";

const writeTheme = async (output: Output) => {
	const preset = await loadPreset("metis-dark");
	return buildArtifact({ output, preset });
};

Go

theme.go 
type TokenRole string

func Resolve(role TokenRole, palette Palette) string {
	if role == "statement" {
		return palette.Amber
	}
	return palette.Foreground
}

Swift

Theme.swift 
struct MetisTheme {
	let name: String
	let syntax: [String: Color]

	func color(for role: String) -> Color {
		syntax[role] ?? syntax["trivial"]!
	}
}

Rust

theme.rs 
pub fn accent(role: &str, theme: &Theme) -> Color {
	match role {
		"string" => theme.mint,
		"number" => theme.yellow,
		_ => theme.foreground,
	}
}
